cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serialization LANGUAGES CXX)
endif()

set(WIRE_BACKEND "zpp" CACHE STRING "Fixed codec backend: zpp or pfr")
set_property(CACHE WIRE_BACKEND PROPERTY STRINGS zpp pfr)
if(NOT TARGET serialization)
  add_library(serialization INTERFACE)
  add_library(LibHT::serialization ALIAS serialization)
  target_compile_features(serialization INTERFACE cxx_std_20)
  target_include_directories(serialization INTERFACE "${CMAKE_CURRENT_LIST_DIR}/include")
  if(WIRE_BACKEND STREQUAL "zpp")
    target_include_directories(serialization SYSTEM INTERFACE "${CMAKE_CURRENT_LIST_DIR}/third_party/zpp")
    target_compile_definitions(serialization INTERFACE WIRE_BACKEND_ZPP=1)
  elseif(WIRE_BACKEND STREQUAL "pfr")
    target_include_directories(serialization SYSTEM INTERFACE "${CMAKE_CURRENT_LIST_DIR}/third_party/pfr/include")
    target_compile_definitions(serialization INTERFACE
      WIRE_BACKEND_PFR=1 BOOST_PFR_USE_CPP17=1 BOOST_PFR_USE_CPP26=0 BOOST_PFR_USE_LOOPHOLE=0)
  else()
    message(FATAL_ERROR "WIRE_BACKEND must be zpp or pfr")
  endif()
endif()

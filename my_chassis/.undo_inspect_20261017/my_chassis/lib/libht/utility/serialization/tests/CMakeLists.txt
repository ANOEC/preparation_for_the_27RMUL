cmake_minimum_required(VERSION 3.22)
project(wire_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "Probe language standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_subdirectory(.. serialization)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(wire_tests codec_tests.cpp)
target_link_libraries(wire_tests PRIVATE LibHT::serialization GTest::gtest_main)
target_compile_options(wire_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(wire_example usage_example.cpp)
target_link_libraries(wire_example PRIVATE LibHT::serialization)
add_library(wire_compile_probe OBJECT compile_probe.cpp)
target_link_libraries(wire_compile_probe PRIVATE LibHT::serialization)
enable_testing()
include(GoogleTest)
gtest_discover_tests(wire_tests)
add_test(NAME wire_example COMMAND wire_example)
add_test(NAME rejected_types COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/compile_checks.py"
  --compiler "${CMAKE_CXX_COMPILER}" --backend "${WIRE_BACKEND}")
find_program(WIRE_ARM_COMPILER arm-none-eabi-g++)
if(WIRE_ARM_COMPILER)
  add_test(NAME arm_compile COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/compile_checks.py"
    --compiler "${WIRE_ARM_COMPILER}" --backend "${WIRE_BACKEND}" --arm)
endif()

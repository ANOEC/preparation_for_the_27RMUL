cmake_minimum_required(VERSION 3.22)
project(wire_benchmarks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(.. serialization)
if(CMAKE_CROSSCOMPILING)
  add_library(wire_dwt_probe OBJECT resource_probe.cpp dwt_probe.cpp)
  target_link_libraries(wire_dwt_probe PRIVATE LibHT::serialization)
  target_compile_options(wire_dwt_probe PRIVATE -O2 -fno-exceptions -fno-rtti -fstack-usage)
else()
  add_executable(wire_host_benchmark host_benchmark.cpp)
  target_link_libraries(wire_host_benchmark PRIVATE LibHT::serialization)
  target_compile_options(wire_host_benchmark PRIVATE -O2 -fno-exceptions -fno-rtti)
endif()
